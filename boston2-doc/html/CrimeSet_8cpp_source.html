22string FormatAsComment(const string &comment, char commentCharacter){
23 size_t start = 0, end;
24 string formatedComment;
25 if (comment.size() > 0) { // If comment has at least one character
26 while ((end = comment.find('\n', start)) != std::string::npos) {
27 formatedComment += commentCharacter + comment.substr(start, end - start) + "\n";
28 start = end + 1;
29 }
30 if(start<comment.size()){ // This happens when last line of comment does not have \n
31 formatedComment += commentCharacter + comment.substr(start, comment.size() - start) + "\n";
32 }
33 }
34 return formatedComment;
35}

379int PosMinArrayCrimes(Crime array[], int initialPos, int finalPos) {
380 // Set the first value as the min value.
381
382 Crime min = array[initialPos];
383 int posMin = initialPos;
384
385 // If there is another value smaller than the real min, then, update
386 // min with the new minimum.
387
388 for (int pos = initialPos; pos <= finalPos ; pos++) {
389
390
391 if (min.IsSmaller(array[pos])) {
392 min = array[pos];
393 posMin = pos;
394 }
395 }
396 if (initialPos > finalPos) posMin = -1;
397
