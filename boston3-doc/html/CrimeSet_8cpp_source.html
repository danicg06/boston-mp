258int CrimeSet::findCrime(const Crime & crime, int initialPos, int finalPos) const {
259 int result = -1;
260
261 bool found = false;
262 int i = initialPos;
263 while (i <= finalPos && !found) { // Search for a crime with the same ID in the array
264 if (_crimes[i].getId() == crime.getId())
265 found = true;
266 else
267 i++;
268 }
269
270 if (found)
271 result = i;
272
273 return result;

280int CrimeSet::posMinArrayCrimes(int initialPos, int finalPos) {
281 // Set the first value as the min value
282 Crime min = _crimes[initialPos];
283 int posMin = initialPos;
284
285 // If there is another value smaller than min, then update min with the
286 // new minimum
287 for (int pos = initialPos+1; pos <= finalPos ; pos++) {
288 if (min.IsSmaller(_crimes[pos])) {
289 min = _crimes[pos];
290 posMin = pos;
291 }
292 }
293 if (initialPos > finalPos) posMin = -1;
294
295 return posMin;
296}

311string FormatAsComment(const string &comment, char commentCharacter) {
312 size_t start = 0, end;
313 string formatedComment;
314 if (comment.size() > 0) { // If comment has at least one character
315 while ((end = comment.find('\n', start)) != std::string::npos) {
316 formatedComment += commentCharacter + comment.substr(start, end - start) + "\n";
317 start = end + 1;
318 }
319 if(start<comment.size()){ // This happens when last line of comment does not have \n
320 formatedComment += commentCharacter + comment.substr(start, comment.size() - start) + "\n";
321 }
322 }
323 return formatedComment;
