168void CrimeCounter::saveAsPPMTextImage(const std::string &fileName,
169 const ColorPalette &palette, const std::string &comment) const{
170
171 const int MAX_COLOR = 255;
172 const string FORMAT = "P3";
173 const string SEPARATOR_FIELDS = "\n";
174 const string SEPARATOR = " ";
175
176 ofstream file(fileName);
177
178 if (file) {
179
180 int colorIndex = 0;
181
182 // Compute maxFrequency as the maximum frequency in the frequency matrix
183 int maxFrequency = getMaxFrequency();
184 // Save heading and comments into the ppm file
185 file << FORMAT + SEPARATOR_FIELDS;
186 file << comment;
187 file << _nCols << SEPARATOR << _nRows << SEPARATOR_FIELDS
188 << MAX_COLOR << SEPARATOR_FIELDS;
189 // For every cell in the frequency matrix
190 for (int i=0; i < _nRows; i++ ){
191 for (int j=0; j < _nCols; j++) {
192 colorIndex = GetIndexPalette(palette,
193 log(_frequency[i][j]+1), log(maxFrequency+1));
194 // Save into the ppm file the RGB components of the color at

196 file << palette.getColor(colorIndex).toString() << SEPARATOR;
197
198 if (!file) {
199 file.close();

222void CrimeCounter::getRowColumn(const Coordinates &coordinates, int& row,
223 int& column) const {
224 if (!coordinates.isValid() ||
225 !coordinates.isInsideArea(_bottomLeftCoord, _topRightCoord)) {
226 throw std::invalid_argument(
227 string("void CrimeCounter::getRowColumn((const Coordinates &loc, int& row, int& column) const: ") +
228 coordinates.toString() + " is not a valid location or is outside the target area");
229 }
230 Coordinates targetArea = this->_bottomLeftCoord.length(this->_topRightCoord);
231 float cellHeight = targetArea.getLatitude()/_nRows;
232 float cellWidth = targetArea.getLongitude()/_nCols;
233 Coordinates delta = _bottomLeftCoord.length(coordinates);
234
235 row = _nRows - (delta.getLatitude() / cellHeight);
236 column = (delta.getLongitude() / cellWidth);
237 if (row == _nRows) row = _nRows-1;
238 if (column == _nCols) column = _nCols-1;
239}
