149void Crime::set(const std::string & line) {
150 const int NFIELDS = 11;
151 string data; //piece of data detached from line;
152 size_t pos = 0;
153 size_t posn = 0; // aux for the beginning of the fields
154
155 for (int nfield = 0; nfield < NFIELDS - 1 && posn != string::npos; nfield++) {
156 posn = line.find(',', pos); // beginning of the next field
157 data = line.substr(pos, posn - pos);
158 pos = posn + 1;
159
160 switch (nfield) {
161 case 0: // counter
162 setCounter(stoi(data));
163 break;
164 case 1: // ID
165 setId(data);
166 break;
167 case 2: // code
168 setCode(data);
169 break;
170 case 3: // group
171 setGroup(data);
172 break;
173 case 4: // desc
174 setDescription(data);
175 break;
176 case 5: // District alphaNumeric
177 setDistrict(data);
178 break;
179 case 6: // areaReport
180 setAreaReport(data);
181 break;
182 case 7: // shooting
183 setShooting(stoi(data));
184 break;
185 case 8: // date time
186 setDateTime(data);
187 break;
188 case 9: // street
189 setStreet(data);
190 break;
191 }
192
193 }
194
195 // location
196 data = line.substr(pos, line.length() - pos + 1);
197 setLocation(Coordinates (data));
198
199 normalize();
200
201} //end of set()

203void Trim(string & myString) {
204 const string SEPARATORS = " \t";
205 size_t pos_first_char = myString.find_first_not_of(SEPARATORS);
206 size_t pos_last_char = myString.find_last_not_of(SEPARATORS);
207
208 if (pos_last_char != string::npos) { // If there is at least one character
209 // that is not an space
210 myString.erase(0, pos_first_char);
211 myString.erase(pos_last_char - pos_first_char + 1);
212 }
213 else // If all characters are spaces
214 myString.clear();
215}
